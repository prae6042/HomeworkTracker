📌 ขั้นตอนการติดตั้งง่ายๆ ใน 3 นาที:

  1. เปิด Google Drive หรือ Google Sheets สร้าง Sheet ใหม่
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้วนำโค้ดด้านล่างนี้ไปวางแทนที่
  4. กดปุ่ม ทำให้ใช้งานได้อย่างเป็นทางการ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment)
  5. เลือกประเภทเป็น เว็บแอป (Web app) | สิทธิ์เข้าถึง (Who has access): เลือก "ทุกคน (Anyone)"
  6. คัดลอก Web App URL นำมาใส่ในเมนูตั้งค่า บนเว็บนี้

// ==========================================
// Google Apps Script Backend for Homework Tracker Pro
// ==========================================

const SHEET_NAME = 'HomeworkData';
const DRIVE_FOLDER_NAME = 'Homework_Attachments';

function doGet(e) {
  const action = e.parameter.action || 'read';
  if (action === 'read') {
    return handleRead();
  }
  return responseJSON({ status: 'error', message: 'คำสั่งไม่ถูกต้อง' });
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;
    
    if (action === 'create') return handleCreate(data);
    if (action === 'updateStatus') return handleUpdateStatus(data);
    if (action === 'delete') return handleDelete(data);
    
    return responseJSON({ status: 'error', message: 'คำสั่งไม่ถูกต้อง' });
  } catch (err) {
    return responseJSON({ status: 'error', message: err.toString() });
  }
}

// 1. อ่านข้อมูลทั้งหมด
function handleRead() {
  const sheet = getOrCreateSheet();
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return responseJSON({ status: 'success', data: [] });
  
  const headers = data[0];
  const rows = data.slice(1);
  const result = rows.map(row => {
    let item = {};
    headers.forEach((header, index) => {
      item[header] = row[index];
    });
    return item;
  });
  
  return responseJSON({ status: 'success', data: result });
}

// 2. สร้างการบ้านใหม่ + บันทึกรูปไป Google Drive
function handleCreate(data) {
  const sheet = getOrCreateSheet();
  let imageUrl = '';
  let fileId = '';
  
  // อัปโหลดรูปภาพเข้า Google Drive ถ้ามีส่งมา
  if (data.imageData && data.imageName) {
    const folder = getOrCreateFolder(DRIVE_FOLDER_NAME);
    const base64Data = data.imageData.split(',')[1] || data.imageData;
    const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), data.imageType || 'image/jpeg', data.imageName);
    const file = folder.createFile(blob);
    file.setSharing(SpreadsheetApp.Access.ANYONE_WITH_LINK, SpreadsheetApp.Permission.VIEW);
    fileId = file.getId();
    imageUrl = 'https://lh3.googleusercontent.com/d/' + fileId;
  }

  const newId = 'HW_' + Date.now();
  const createdAt = new Date().toISOString();

  sheet.appendRow([
    newId,
    data.subject,
    data.title,
    data.description || '',
    data.assignDate || '',
    data.dueDate,
    data.priority || 'ปกติ',
    'pending',
    imageUrl,
    fileId,
    createdAt
  ]);

  return responseJSON({ 
    status: 'success', 
    id: newId, 
    imageUrl: imageUrl, 
    message: 'บันทึกเรียบร้อยแล้ว' 
  });
}

// 3. อัปเดตสถานะการบ้าน
function handleUpdateStatus(data) {
  const sheet = getOrCreateSheet();
  const values = sheet.getDataRange().getValues();
  
  for (let i = 1; i < values.length; i++) {
    if (values[i][0] === data.id) {
      sheet.getRange(i + 1, 8).setValue(data.status); // Column 8 คือ Status
      return responseJSON({ status: 'success', message: 'อัปเดตสถานะเรียบร้อย' });
    }
  }
  return responseJSON({ status: 'error', message: 'ไม่พบรายการที่ต้องการ' });
}

// 4. ลบการบ้าน
function handleDelete(data) {
  const sheet = getOrCreateSheet();
  const values = sheet.getDataRange().getValues();
  
  for (let i = 1; i < values.length; i++) {
    if (values[i][0] === data.id) {
      sheet.deleteRow(i + 1);
      return responseJSON({ status: 'success', message: 'ลบรายการเรียบร้อย' });
    }
  }
  return responseJSON({ status: 'error', message: 'ไม่พบรายการที่ต้องการ' });
}

// Helper: สร้าง Sheet พร้อมหัวตารางอัตโนมัติ
function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      'id', 
      'subject', 
      'title', 
      'description', 
      'assignDate', 
      'dueDate', 
      'priority', 
      'status', 
      'imageUrl', 
      'fileId', 
      'createdAt'
    ]);
    sheet.getRange(1, 1, 1, 11).setFontWeight('bold').setBackground('#4f46e5').setFontColor('#ffffff');
  }
  return sheet;
}

// Helper: สร้าง Folder ใน Google Drive อัตโนมัติ
function getOrCreateFolder(folderName) {
  const folders = DriveApp.getFoldersByName(folderName);
  if (folders.hasNext()) return folders.next();
  return DriveApp.createFolder(folderName);
}

// Helper: คืนค่า JSON สำหรับ CORS
function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
